137 // right time zone

140 auto dp = date::floor<date::days>(tp);

142 auto ymd = date::year_month_day{dp};

144
145 time_parts.clear();
146
147 // push year/month/day

155 time_parts.push_back(time.seconds().count());
156 time_parts.push_back(time.subseconds().count());

175 // format time point to string, chars in <> are keywords.
176 // in: <year>-<mon>-<day> <hour>:<min>:<sec>.<mili>
177 // out: 2022-10-08 13:53:07.230
178 // or
179 // in: [<day>/<mon>/<year> <hour>:<min>:<sec>.<mili>]
180 // out: [08/10/2022 13:53:07.230]
181 inline std::string time_format(system_clock::time_point tp, std::string template_str = "<year>-<mon>-<day> <hour>:<min>:<sec>.<mili>", int time_zone = 8) {
182 // right time zone
183 tp = tp + std::chrono::hours{time_zone};
184
185 // 7 parts
186 std::vector<int> time_parts;
187 time_split(tp, time_parts, 0);
188
189 auto time_str = std::regex_replace(template_str, std::regex("<year>"), std::to_string(time_parts[0]));
190 time_str = std::regex_replace(time_str, std::regex("<mon>"), set_width_and_fill(std::to_string(time_parts[1]), 2));
191 time_str = std::regex_replace(time_str, std::regex("<day>"), set_width_and_fill(std::to_string(time_parts[2]), 2));
192 time_str = std::regex_replace(time_str, std::regex("<hour>"), set_width_and_fill(std::to_string(time_parts[3]), 2));
193 time_str = std::regex_replace(time_str, std::regex("<min>"), set_width_and_fill(std::to_string(time_parts[4]), 2));
194 time_str = std::regex_replace(time_str, std::regex("<sec>"), set_width_and_fill(std::to_string(time_parts[5]), 2));
195 time_str = std::regex_replace(time_str, std::regex("<mili>"), set_width_and_fill(std::to_string(time_parts[6]), 3));
196
197 return time_str;
198 }
