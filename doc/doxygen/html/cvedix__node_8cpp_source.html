15  void cvedix_node::handle_run() {
16  // cache for batch handling if need
17  std::vector<std::shared_ptr<cvedix_objects::cvedix_frame_meta>> frame_meta_batch_cache;
18  while (alive) {
19  // wait for producer, make sure in_queue is not empty.
20  this->in_queue_semaphore.wait();
21 
22  CVEDIX_DEBUG(cvedix_utils::string_format("[%s] before handling meta, in_queue.size()==>%d", node_name.c_str(), in_queue.size()));
23  auto in_meta = this->in_queue.front();
24 
25  // dead flag
26  if (in_meta == nullptr) {
27  continue;
28  }
29 
30  // handling hooker activated if need
31  invoke_meta_handling_hooker(node_name, in_queue.size(), in_meta);
32 
33  std::shared_ptr<cvedix_objects::cvedix_meta> out_meta;
34  auto batch_complete = false;
35 
36  // call handlers
37  if (in_meta->meta_type == cvedix_objects::cvedix_meta_type::CONTROL) {
38  auto meta_2_handle = std::dynamic_pointer_cast<cvedix_objects::cvedix_control_meta>(in_meta);
39  out_meta = this->handle_control_meta(meta_2_handle);
40  }
41  else if (in_meta->meta_type == cvedix_objects::cvedix_meta_type::FRAME) {
42  auto meta_2_handle = std::dynamic_pointer_cast<cvedix_objects::cvedix_frame_meta>(in_meta);
43  // one by one
44  if (frame_meta_handle_batch == 1) {
45  out_meta = this->handle_frame_meta(meta_2_handle);
46  }
47  else {
48  // batch by batch
49  frame_meta_batch_cache.push_back(meta_2_handle);
50  if (frame_meta_batch_cache.size() >= frame_meta_handle_batch) {
51  // cache complete
52  this->handle_frame_meta(frame_meta_batch_cache);
53  batch_complete = true;
54  }
55  else {
56  // cache not complete, do nothing
57  CVEDIX_DEBUG(cvedix_utils::string_format("[%s] handle meta with batch, frame_meta_batch_cache.size()==>%d", node_name.c_str(), frame_meta_batch_cache.size()));
58  }
59  }
60  }
61  else {
62  throw "invalid meta type!";
63  }
64  this->in_queue.pop();
65  CVEDIX_DEBUG(cvedix_utils::string_format("[%s] after handling meta, in_queue.size()==>%d", node_name.c_str(), in_queue.size()));
66 
67  // one by one mode
68  // return nullptr means do not push it to next nodes(such as in des nodes).
69  if (out_meta != nullptr && node_type() != cvedix_node_type::DES) {
70  CVEDIX_DEBUG(cvedix_utils::string_format("[%s] before handling meta, out_queue.size()==>%d", node_name.c_str(), out_queue.size()));
71  this->out_queue.push(out_meta);
72 
73  // handled hooker activated if need
74  invoke_meta_handled_hooker(node_name, out_queue.size(), out_meta);
75 
76  // notify consumer of out_queue
77  this->out_queue_semaphore.signal();
78  CVEDIX_DEBUG(cvedix_utils::string_format("[%s] after handling meta, out_queue.size()==>%d", node_name.c_str(), out_queue.size()));
79  }
80 
81  // batch by batch mode
82  if (batch_complete && node_type() != cvedix_node_type::DES) {
83  // push to out_queue one by one
84  for (auto& i: frame_meta_batch_cache) {
85  CVEDIX_DEBUG(cvedix_utils::string_format("[%s] before handling meta, out_queue.size()==>%d", node_name.c_str(), out_queue.size()));
86  this->out_queue.push(i);
87 
88  // handled hooker activated if need
89  invoke_meta_handled_hooker(node_name, out_queue.size(), i);
90 
91  // notify consumer of out_queue
92  this->out_queue_semaphore.signal();
93  CVEDIX_DEBUG(cvedix_utils::string_format("[%s] after handling meta, out_queue.size()==>%d", node_name.c_str(), out_queue.size()));
94  }
95  // clean cache for the next batch
96  frame_meta_batch_cache.clear();
97  }
98  }
99  // send dead flag for dispatch_thread
100  this->out_queue.push(nullptr);
101  this->out_queue_semaphore.signal();
102  }

109

122  this->out_queue.pop();
