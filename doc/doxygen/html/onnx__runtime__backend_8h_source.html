72 std::vector<float> extractEmbedding(const cv::Mat& aligned_face) override {
73 if (!ready_) {
74 return std::vector<float>(embedding_dim_, 0.0f);
75 }
76
77 // Preprocess: normalize to [-1, 1]
78 cv::Mat rgb;
79 if (aligned_face.channels() == 3) {
80 cv::cvtColor(aligned_face, rgb, cv::COLOR_BGR2RGB);
81 } else {
82 rgb = aligned_face;
83 }
84
85 cv::Mat resized;
86 cv::resize(rgb, resized, cv::Size(input_width_, input_height_));
87
88 cv::Mat float_img;
89 resized.convertTo(float_img, CV_32FC3, 1.0f / 127.5f, -1.0f);
90
91 // Prepare input tensor (NCHW format)
92 std::vector<float> input_tensor(1 * 3 * input_height_ * input_width_);
93 for (int c = 0; c < 3; c++) {
94 for (int h = 0; h < input_height_; h++) {
95 for (int w = 0; w < input_width_; w++) {
96 input_tensor[c * input_height_ * input_width_ + h * input_width_ + w] =
97 float_img.ptr<cv::Vec3f>(h)[w][c];
98 }
99 }
100 }
101
102 // Run inference
103 std::vector<int64_t> input_shape = {1, 3, input_height_, input_width_};
104 auto memory_info = Ort::MemoryInfo::CreateCpu(OrtArenaAllocator, OrtMemTypeDefault);
105 Ort::Value input_tensor_ort = Ort::Value::CreateTensor<float>(
106 memory_info, input_tensor.data(), input_tensor.size(),
107 input_shape.data(), input_shape.size()
108 );
109
110 auto output_tensors = session_->Run(
111 Ort::RunOptions{nullptr},
112 input_names_.data(), &input_tensor_ort, 1,
113 output_names_.data(), output_names_.size()
114 );
115
116 // Extract embedding
117 float* output_data = output_tensors[0].GetTensorMutableData<float>();
118 std::vector<float> embedding(embedding_dim_);
119
120 float norm = 0.0f;
121 for (int i = 0; i < embedding_dim_; i++) {
122 embedding[i] = output_data[i];
123 norm += embedding[i] * embedding[i];
124 }
125 norm = std::sqrt(norm);
126 if (norm > 1e-6f) {
127 for (float& v : embedding) v /= norm;
128 }
129
130 return embedding;
131 }
