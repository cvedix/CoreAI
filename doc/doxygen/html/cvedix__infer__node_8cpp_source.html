79  void cvedix_infer_node::infer(const cv::Mat& blob_to_infer, std::vector<cv::Mat>& raw_outputs) {
80  // blob_to_infer is a 4D matrix
81  // the first dim is number of batch
82  assert(blob_to_infer.dims == 4);
83  assert(!net.empty());
84 
85  auto number_of_batch = blob_to_infer.size[0];
86  if (number_of_batch <= batch_size) {
87  // infer one time directly
88  net.setInput(blob_to_infer);
89  net.forward(raw_outputs, net.getUnconnectedOutLayersNames());
90  }
91  else {
92  // infer more times
93  int b_size[] = {batch_size, blob_to_infer.size[1], blob_to_infer.size[2], blob_to_infer.size[3]};
94  auto times = (number_of_batch % batch_size) == 0 ? (number_of_batch / batch_size) : (number_of_batch / batch_size + 1);
95  for (int i = 0; i < times; i++) {
96  // split to small piece
97  int i_hwc[] = {i * batch_size, 0, 0, 0}; // 4D
98  auto ptr = blob_to_infer.ptr(i_hwc);
99  cv::Mat b_blob(4, b_size, CV_32F, (void*)ptr);
100  std::vector<cv::Mat> b_outputs;
101 
102  net.setInput(b_blob);
103  net.forward(b_outputs, net.getUnconnectedOutLayersNames());
104 
105  // first time, initialize it
106  if (raw_outputs.size() == 0) {
107  // scan multi heads of output
108  for (int j = 0; j < b_outputs.size(); j++) {
109  if (batch_size == 1) {
110  // keep dims as usual, but change size[0] == number_of_batch
111  if (b_outputs[j].dims <= 2 && b_outputs[j].rows == 1) {
112  raw_outputs.push_back(cv::Mat(2, std::vector<int>{number_of_batch, b_outputs[j].cols}.data(), CV_32F));
113  }
114  else {
115  // dims add 1, and set size[0] == number_of_batch
116  std::vector<int> t_size;
117  t_size.push_back(number_of_batch);
118  for (int s = 0; s < b_outputs[j].dims; s++) {
119  t_size.push_back(b_outputs[j].size[s]);
120  }
121  raw_outputs.push_back(cv::Mat(b_outputs[j].dims + 1, t_size.data(), CV_32F));
122  }
123  }
124  else {
125  // kepp dims as usual, but change size[0] == number_of_batch
126  std::vector<int> t_size;
127  t_size.push_back(number_of_batch);
128 
129  // start from 1
130  for (int s = 1; s < b_outputs[j].dims; s++) {
131  /* code */
132  t_size.push_back(b_outputs[j].size[s]);
133  }
134  raw_outputs.push_back(cv::Mat(b_outputs[j].dims, t_size.data(), CV_32F));
135  }
136  }
137  }
138 
139  assert(raw_outputs.size() == b_outputs.size());
140  // merge data directly
141  for (int j = 0; j < b_outputs.size(); j++) {
142  auto& des = raw_outputs[j];
143  auto& src = b_outputs[j];
144 
145  std::vector<int> t_size;
146  auto s_dims_n = src.dims <= 2 ? 2 : src.dims;
147  for (int s = 0; s < s_dims_n; s++) {
148  t_size.push_back(src.size[s]);
149  }
150 
151  auto ptr = des.ptr(i * batch_size);
152  cv::Mat tmp(s_dims_n, t_size.data(), CV_32F, (void*)ptr);
153 
154  src.copyTo(tmp);
155  }
156  }
157  }
158  }

160  // Helper function to transpose NCHW -> NHWC (replacement for cv::transposeND in OpenCV < 4.8)
161  static void transposeNCHWtoNHWC(const cv::Mat& src, cv::Mat& dst) {
162  // src: [N, C, H, W] -> dst: [N, H, W, C]
163  if (src.dims != 4) {
164  dst = src.clone();
165  return;
166  }
167 
168  int n = src.size[0];
169  int c = src.size[1];
170  int h = src.size[2];
171  int w = src.size[3];
172 
173  // Create output with shape [N, H, W, C]
174  std::vector<int> new_shape = {n, h, w, c};
175  dst = cv::Mat(4, new_shape.data(), src.type());
176 
177  // Copy data with permutation
178  const float* src_data = (const float*)src.data;
179  float* dst_data = (float*)dst.data;
180 
181  int src_step_c = h * w;
182  int src_step_h = w;
183  int dst_step_h = w * c;
184  int dst_step_w = c;
185 
186  for (int ni = 0; ni < n; ni++) {
187  for (int hi = 0; hi < h; hi++) {
188  for (int wi = 0; wi < w; wi++) {
189  for (int ci = 0; ci < c; ci++) {
190  int src_idx = ni * (c * h * w) + ci * (h * w) + hi * w + wi;
191  int dst_idx = ni * (h * w * c) + hi * (w * c) + wi * c + ci;
192  dst_data[dst_idx] = src_data[src_idx];
193  }
194  }
195  }
196  }
197  }

215  void cvedix_infer_node::run_infer_combinations(const std::vector<std::shared_ptr<cvedix_objects::cvedix_frame_meta>>& frame_meta_with_batch) {
216  /*
217  * call logic by default:
218  * frame_meta_with_batch -> mats_to_infer -> blob_to_infer -> raw_outputs -> frame_meta_with_batch
219  */
220  std::vector<cv::Mat> mats_to_infer;
221  // 4D matrix
222  cv::Mat blob_to_infer;
223  // multi heads of output in network, raw matrix output which need to be parsed by users.
224  std::vector<cv::Mat> raw_outputs;
225 
226  // start
227  auto start_time = std::chrono::system_clock::now();
228  // 1st step, prepare
229  prepare(frame_meta_with_batch, mats_to_infer);
230  auto prepare_time = std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::system_clock::now() - start_time);
231 
232  // nothing to infer
233  if (mats_to_infer.size() == 0) {
234  return;
235  }
236 
237  start_time = std::chrono::system_clock::now();
238  // 2nd step, preprocess
239  preprocess(mats_to_infer, blob_to_infer);
240  auto preprocess_time = std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::system_clock::now() - start_time);
241 
242  start_time = std::chrono::system_clock::now();
243  // 3rd step, infer
244  infer(blob_to_infer, raw_outputs);
245  auto infer_time = std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::system_clock::now() - start_time);
246 
247  start_time = std::chrono::system_clock::now();
248  // 4th step, postprocess
249  postprocess(raw_outputs, frame_meta_with_batch);
250  auto postprocess_time = std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::system_clock::now() - start_time);
251 
252  // end
253  infer_combinations_time_cost(mats_to_infer.size(), prepare_time.count(), preprocess_time.count(), infer_time.count(), postprocess_time.count());
254  }
