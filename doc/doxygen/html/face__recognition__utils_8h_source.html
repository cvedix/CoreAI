64 std::pair<std::string, float> get_stable_result(int track_id) {
65 auto it = track_buffers_.find(track_id);
66 if (it == track_buffers_.end() || it->second.empty()) {
67 return {"Unknown", 0.0f};
68 }
69
70 const auto& buffer = it->second;
71
72 // Count votes
73 std::map<std::string, int> vote_count;
74 std::map<std::string, float> score_sum;
75
76 for (const auto& entry : buffer) {
77 vote_count[entry.name]++;
78 score_sum[entry.name] += entry.score;
79 }
80
81 // Find winner
82 std::string winner = "Unknown";
83 int max_votes = 0;
84 float winner_score = 0.0f;
85
86 for (const auto& [name, count] : vote_count) {
87 if (count > max_votes) {
88 max_votes = count;
89 winner = name;
90 winner_score = score_sum[name] / count; // Average score
91 }
92 }
93
94 // Check majority threshold
95 float vote_ratio = static_cast<float>(max_votes) / buffer.size();
96 if (vote_ratio >= majority_threshold_) {
97 return {winner, winner_score};
98 }
99
100 return {"Unknown", 0.0f}; // No majority
101 }

156inline std::vector<float> average_embeddings(
157 const std::vector<std::vector<float>>& embeddings) {
158
159 if (embeddings.empty()) return {};
160 if (embeddings.size() == 1) return embeddings[0];
161
162 size_t dim = embeddings[0].size();
163 std::vector<float> result(dim, 0.0f);
164
165 for (const auto& emb : embeddings) {
166 for (size_t i = 0; i < dim && i < emb.size(); i++) {
167 result[i] += emb[i];
168 }
169 }
170
171 float n = static_cast<float>(embeddings.size());
172 for (float& val : result) {
173 val /= n;
174 }
175
176 // L2 normalize the averaged embedding
177 float norm = 0.0f;
178 for (float val : result) norm += val * val;
179 norm = std::sqrt(norm);
180 if (norm > 1e-6f) {
181 for (float& val : result) val /= norm;
182 }
183
184 return result;
185}
