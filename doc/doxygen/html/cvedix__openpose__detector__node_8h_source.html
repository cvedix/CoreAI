82  void getKeyPoints(cv::Mat& probMap, double threshold, std::vector<KeyPoint>& keyPoints) {
83  cv::Mat smoothProbMap;
84  cv::GaussianBlur(probMap, smoothProbMap, cv::Size( 3, 3 ), 0, 0);
85 
86  cv::Mat maskedProbMap;
87  cv::threshold(smoothProbMap, maskedProbMap, threshold, 255, cv::THRESH_BINARY);
88 
89  maskedProbMap.convertTo(maskedProbMap, CV_8U, 1);
90 
91  std::vector<std::vector<cv::Point> > contours;
92  cv::findContours(maskedProbMap, contours, cv::RETR_TREE, cv::CHAIN_APPROX_SIMPLE);
93 
94  for(int i = 0; i < contours.size();++i) {
95  cv::Mat blobMask = cv::Mat::zeros(smoothProbMap.rows, smoothProbMap.cols, smoothProbMap.type());
96  cv::fillConvexPoly(blobMask, contours[i], cv::Scalar(1));
97 
98  double maxVal;
99  cv::Point maxLoc;
100 
101  cv::minMaxLoc(smoothProbMap.mul(blobMask), 0, &maxVal, 0, &maxLoc);
102  keyPoints.push_back(KeyPoint(maxLoc, probMap.at<float>(maxLoc.y, maxLoc.x)));
103  }
104  }

122  void populateInterpPoints(const cv::Point& a,const cv::Point& b,int numPoints,std::vector<cv::Point>& interpCoords){
123  float xStep = ((float)(b.x - a.x)) / (float)(numPoints - 1);
124  float yStep = ((float)(b.y - a.y)) / (float)(numPoints - 1);
125  interpCoords.push_back(a);
126 
127  for(int i = 1; i< numPoints - 1; ++i) {
128  interpCoords.push_back(cv::Point(a.x + xStep * i, a.y + yStep * i));
129  }
130  interpCoords.push_back(b);

133  void getValidPairs(const std::vector<cv::Mat>& netOutputParts,
134  const std::vector<std::vector<KeyPoint>>& detectedKeypoints,
135  std::vector<std::vector<ValidPair>>& validPairs,
136  std::set<int>& invalidPairs) {
137 
138  int nInterpSamples = 10;
139  float confTh = 0.7;
140 
141  for(int k = 0; k < mapIdxes_map.at(type).size(); ++k) {
142  //A->B constitute a limb
143  cv::Mat pafA = netOutputParts[mapIdxes_map.at(type)[k].first + points_map.at(type) + 1];
144  cv::Mat pafB = netOutputParts[mapIdxes_map.at(type)[k].second + points_map.at(type) + 1];
145 
146  //Find the keypoints for the first and second limb
147  const std::vector<KeyPoint>& candA = detectedKeypoints[posePairs_map.at(type)[k].first];
148  const std::vector<KeyPoint>& candB = detectedKeypoints[posePairs_map.at(type)[k].second];
149 
150  int nA = candA.size();
151  int nB = candB.size();
152 
153  /*
154  # If keypoints for the joint-pair is detected
155  # check every joint in candA with every joint in candB
156  # Calculate the distance vector between the two joints
157  # Find the PAF values at a set of interpolated points between the joints
158  # Use the above formula to compute a score to mark the connection valid
159  */
160 
161  if(nA != 0 && nB != 0){
162  std::vector<ValidPair> localValidPairs;
163 
164  for(int i = 0; i< nA; ++i) {
165  int maxJ = -1;
166  float maxScore = -1;
167  bool found = false;
168 
169  for(int j = 0; j < nB; ++j) {
170  std::pair<float, float> distance(candB[j].point.x - candA[i].point.x, candB[j].point.y - candA[i].point.y);
171 
172  float norm = std::sqrt(distance.first * distance.first + distance.second * distance.second);
173 
174  if(!norm) {
175  continue;
176  }
177 
178  distance.first /= norm;
179  distance.second /= norm;
180 
181  //Find p(u)
182  std::vector<cv::Point> interpCoords;
183  populateInterpPoints(candA[i].point, candB[j].point, nInterpSamples, interpCoords);
184  //Find L(p(u))
185  std::vector<std::pair<float, float>> pafInterp;
186  for(int l = 0; l < interpCoords.size();++l) {
187  pafInterp.push_back(
188  std::pair<float,float>(
189  pafA.at<float>(interpCoords[l].y, interpCoords[l].x),
190  pafB.at<float>(interpCoords[l].y, interpCoords[l].x)
191  ));
192  }
193 
194  std::vector<float> pafScores;
195  float sumOfPafScores = 0;
196  int numOverTh = 0;
197  for(int l = 0; l< pafInterp.size(); ++l){
198  float score = pafInterp[l].first * distance.first + pafInterp[l].second * distance.second;
199  sumOfPafScores += score;
200  if(score > score_threshold){
201  ++numOverTh;
202  }
203 
204  pafScores.push_back(score);
205  }
206 
207  float avgPafScore = sumOfPafScores / ((float)pafInterp.size());
208 
209  if(((float)numOverTh)/((float)nInterpSamples) > confTh){
210  if(avgPafScore > maxScore) {
211  maxJ = j;
212  maxScore = avgPafScore;
213  found = true;
214  }
215  }
216  }/* j */
217 
218  if(found) {
219  localValidPairs.push_back(ValidPair(candA[i].id, candB[maxJ].id, maxScore));
220  }
221 
222  }/* i */
223 
224  validPairs.push_back(localValidPairs);
225 
226  } else {
227  invalidPairs.insert(k);
228  validPairs.push_back(std::vector<ValidPair>());
229  }
230  }/* k */
231  }

233  void getPersonwiseKeypoints(const std::vector<std::vector<ValidPair>>& validPairs,

238  continue;
239  }
240

242 
243  int indexA(posePairs_map.at(type)[k].first);
244  int indexB(posePairs_map.at(type)[k].second);
245 
246  for(int i = 0; i< localValidPairs.size(); ++i) {
247  bool found = false;
248  int personIdx = -1;
249

251  if(indexA < personwiseKeypoints[j].size() &&
252  personwiseKeypoints[j][indexA] == localValidPairs[i].aId) {
253  personIdx = j;
254  found = true;
255  }
256  }/* j */
257

262  std::vector<int> lpkp(std::vector<int>(points_map.at(type), -1));

264  lpkp.at(indexA) = localValidPairs[i].aId;
265  lpkp.at(indexB) = localValidPairs[i].bId;

270  }/* i */
