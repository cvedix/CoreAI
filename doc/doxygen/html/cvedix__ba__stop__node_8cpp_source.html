36  bool cvedix_ba_stop_node::point_in_poly(cvedix_objects::cvedix_point p, std::vector<cvedix_objects::cvedix_point> region) {
37  int i, j, c = 0;
38  int nvert = region.size();
39 
40  for (i = 0, j = nvert-1; i < nvert; j = i++) {
41  if (((region[i].y > p.y) != (region[j].y > p.y)) &&
42  (p.x < (region[j].x - region[i].x) * (p.y - region[i].y)
43  / (region[j].y - region[i].y) + region[i].x))
44  c = !c;
45  }
46  return c;
