60  original_height = video_height;
61

64  }

66  cvedix_stream_info stream_info {channel_index, original_fps, original_width, original_height, to_string()};

69  rtmp_capture >> frame;

77  skip++;

79  }

104  }
105

107  this->out_queue_semaphore.signal();
108  CVEDIX_DEBUG(cvedix_utils::string_format("[%s] after handling meta, out_queue.size()==>%d", node_name.c_str(), out_queue.size()));
109  }
110  }
111 
112  // send dead flag for dispatch_thread
113  this->out_queue.push(nullptr);
