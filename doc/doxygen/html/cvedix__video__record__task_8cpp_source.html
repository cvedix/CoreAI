53 
54  // pop data from cache and write into file
55  while (status == cvedix_record_task_status::STARTED) {
56  // it is a consumer
57  cache_semaphore.wait();

65  cv::Mat frame_data;

67  preprocess(frame_to_record, frame_data);

69

71  if (!video_writer.isOpened()) {
72  assert(video_writer.open(gst, cv::CAP_GSTREAMER, 0, fps, {frame_data.cols, frame_data.rows}));
