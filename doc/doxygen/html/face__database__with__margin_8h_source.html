77 bool load(const std::string& path, bool normalize = true) {
78 std::ifstream db(path);
79 if (!db.is_open()) {
80 CVEDIX_ERROR(cvedix_utils::string_format(
81 "[FaceDB] Failed to open database: %s", path.c_str()));
82 return false;
83 }
84
85 std::string line;
86 int count = 0;
87 int line_num = 0;
88
89 while (std::getline(db, line)) {
90 line_num++;
91 if (line.empty() || line[0] == '#') {
92 continue; // Skip empty lines and comments
93 }
94
95 // Parse name and embedding
96 std::string name;
97 std::vector<float> emb;
98
99 // Try pipe-separated format first (name|emb)
100 size_t pipe_pos = line.find('|');
101 if (pipe_pos != std::string::npos) {
102 name = line.substr(0, pipe_pos);
103 std::string emb_str = line.substr(pipe_pos + 1);
104
105 // Parse comma-separated values
106 std::stringstream ss(emb_str);
107 std::string val;
108 while (std::getline(ss, val, ',')) {
109 try {
110 emb.push_back(std::stof(val));
111 } catch (...) {
112 CVEDIX_ERROR(cvedix_utils::string_format(
113 "[FaceDB] Invalid embedding value at line %d", line_num));
114 return false;
115 }
116 }
117 } else {
118 // Try space-separated format (name emb[0] emb[1] ...)
119 std::stringstream ss(line);
120 ss >> name;
121 float val;
122 while (ss >> val) {
123 emb.push_back(val);
124 }
125 }
126
127 // Validate embedding size
128 if (emb.size() != 512) {
129 CVEDIX_ERROR(cvedix_utils::string_format(
130 "[FaceDB] Invalid embedding size for %s: %d (expected 512) at line %d",
131 name.c_str(), static_cast<int>(emb.size()), line_num));
132 return false;
133 }
134
135 // L2 normalize if requested
136 if (normalize) {
137 float norm = 0.0f;
138 for (float v : emb) {
139 norm += v * v;
140 }
141 norm = std::sqrt(norm);
142 if (norm > 1e-6) {
143 for (float& v : emb) {
144 v /= norm;
145 }
146 }
147 }
148
149 // Add to database
150 if (embeddings.find(name) == embeddings.end()) {
151 embeddings[name] = std::vector<std::vector<float>>();
152 }
153 embeddings[name].push_back(emb);
154 count++;
155 }
156
157 CVEDIX_INFO(cvedix_utils::string_format(
158 "[FaceDB] Loaded %d embeddings for %d people",
159 count, static_cast<int>(embeddings.size())));
160
161 return true;
162 }

170 MatchResult find_match(const std::vector<float>& query_embedding) {
171 total_queries++;
172
173 MatchResult result;
174 result.name = "Unknown";
175 result.score = -1.0f;
176 result.margin = 0.0f;
177 result.confident = false;
178 result.second_name = "Unknown";
179 result.second_score = -1.0f;
180
181 if (embeddings.empty()) {
182 return result;
183 }
184
185 // Find top-2 matches
186 std::vector<std::pair<std::string, float>> scores;
187
188 for (const auto& [name, emb_list] : embeddings) {
189 // For multiple embeddings per person, use the best match
190 float best_score = -1.0f;
191 for (const auto& db_emb : emb_list) {
192 float score = cosine_similarity(query_embedding, db_emb);
193 best_score = std::max(best_score, score);
194 }
195 scores.push_back({name, best_score});
196 }
197
198 // Sort by score descending
199 std::sort(scores.begin(), scores.end(),
200 [](const auto& a, const auto& b) { return a.second > b.second; });
201
202 // Get top-1 and top-2
203 result.name = scores[0].first;
204 result.score = scores[0].second;
205
206 if (scores.size() >= 2) {
207 result.second_name = scores[1].first;
208 result.second_score = scores[1].second;
209 result.margin = result.score - result.second_score;
210 } else {
211 result.margin = result.score; // Only one person in database
212 }
213
214 // Apply filtering logic
215 bool passes_threshold = result.score >= similarity_threshold;
216 bool passes_margin = !use_margin_check || (result.margin >= confidence_margin);
217
218 if (strict_mode) {
219 // Strict: BOTH conditions must pass
220 result.confident = passes_threshold && passes_margin;
221 } else {
222 // Relaxed: Pass if threshold is met, OR if margin is good with lower threshold
223 float relaxed_threshold = similarity_threshold - 0.1f;
224 result.confident = passes_threshold && passes_margin;
225
226 // Alternative: high margin can compensate for slightly lower score
227 if (!result.confident && result.margin >= confidence_margin &&
228 result.score >= relaxed_threshold) {
229 result.confident = true;
230 CVEDIX_DEBUG(cvedix_utils::string_format(
231 "[FaceDB] Accepted with relaxed threshold: score=%.3f (margin=%.3f)",
232 result.score, result.margin));
233 }
234 }
235
236 // Update statistics
237 if (result.confident) {
238 accepted_matches++;
239 } else if (!passes_threshold) {
240 rejected_by_threshold++;
241 } else {
242 rejected_by_margin++;
243 }
244
245 // If not confident, return "Unknown"
246 if (!result.confident) {
247 result.name = "Unknown";
248 }
249
250 CVEDIX_DEBUG(cvedix_utils::string_format(
251 "[FaceDB] Query result: %s (score=%.3f, margin=%.3f, confident=%s)",
252 result.name.c_str(), result.score, result.margin,
253 result.confident ? "YES" : "NO"));
254
255 return result;
256 }

373 float cosine_similarity(const std::vector<float>& a, const std::vector<float>& b) {
374 if (a.size() != b.size()) {
375 return -1.0f;
376 }
377
378 float dot = 0.0f;
379 for (size_t i = 0; i < a.size(); i++) {
380 dot += a[i] * b[i];
381 }
382
383 // If embeddings are L2 normalized, dot product = cosine similarity
384 // Otherwise, need to normalize:
385 // return dot / (norm(a) * norm(b));
386
387 return dot;
388 }
