53  // 'tl': top left point of the bounding box

55  // 'nt': nose tip

78  float y1 = cy - h / 2;

126  auto y = int(nms_faces.at<float>(i, 1));

168  void cvedix_yunet_face_detector_node::generatePriors() {
169  using namespace cv;
170  // Calculate shapes of different scales according to the shape of input image
171  Size feature_map_2nd = {
172  int(int((inputW+1)/2)/2), int(int((inputH+1)/2)/2)
173  };
174  Size feature_map_3rd = {
175  int(feature_map_2nd.width/2), int(feature_map_2nd.height/2)
176  };
177  Size feature_map_4th = {
178  int(feature_map_3rd.width/2), int(feature_map_3rd.height/2)
179  };
180  Size feature_map_5th = {
181  int(feature_map_4th.width/2), int(feature_map_4th.height/2)
182  };
183  Size feature_map_6th = {
184  int(feature_map_5th.width/2), int(feature_map_5th.height/2)
185  };
186 
187  std::vector<Size> feature_map_sizes;
188  feature_map_sizes.push_back(feature_map_3rd);
189  feature_map_sizes.push_back(feature_map_4th);
190  feature_map_sizes.push_back(feature_map_5th);
191  feature_map_sizes.push_back(feature_map_6th);
192 
193  // Fixed params for generating priors
194  const std::vector<std::vector<float>> min_sizes = {
195  {10.0f, 16.0f, 24.0f},
196  {32.0f, 48.0f},
197  {64.0f, 96.0f},
198  {128.0f, 192.0f, 256.0f}
199  };
200  CV_Assert(min_sizes.size() == feature_map_sizes.size()); // just to keep vectors in sync
201  const std::vector<int> steps = { 8, 16, 32, 64 };
202 
203  // Generate priors
204  priors.clear();
205  for (size_t i = 0; i < feature_map_sizes.size(); ++i)
206  {
207  Size feature_map_size = feature_map_sizes[i];
208  std::vector<float> min_size = min_sizes[i];
209 
210  for (int _h = 0; _h < feature_map_size.height; ++_h)
211  {
212  for (int _w = 0; _w < feature_map_size.width; ++_w)
213  {
214  for (size_t j = 0; j < min_size.size(); ++j)
215  {
216  float s_kx = min_size[j] / inputW;
217  float s_ky = min_size[j] / inputH;
218 
219  float cx = (_w + 0.5f) * steps[i] / inputW;
220  float cy = (_h + 0.5f) * steps[i] / inputH;
221 
222  Rect2f prior = { cx, cy, s_kx, s_ky };
223  priors.push_back(prior);
224  }
225  }
226  }
227  }
228  }
