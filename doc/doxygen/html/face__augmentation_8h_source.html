161 cv::Mat apply_glasses(const cv::Mat& aligned_face) {
162 if (!has_glasses_) {
163 return aligned_face.clone();
164 }
165
166 cv::Mat result = aligned_face.clone();
167
168 // Calculate eye center and width
169 cv::Point2f eye_center((ref_left_eye_.x + ref_right_eye_.x) / 2.0f,
170 (ref_left_eye_.y + ref_right_eye_.y) / 2.0f);
171 float eye_width = std::abs(ref_right_eye_.x - ref_left_eye_.x);
172
173 // Scale glasses to fit eye width (glasses should be ~1.8x eye width)
174 float target_width = eye_width * 1.8f;
175 float scale = target_width / glasses_template_.cols;
176
177 cv::Mat resized_glasses;
178 cv::resize(glasses_template_, resized_glasses, cv::Size(), scale, scale, cv::INTER_LINEAR);
179
180 // Position glasses centered on eyes
181 int x = static_cast<int>(eye_center.x - resized_glasses.cols / 2.0f);
182 int y = static_cast<int>(eye_center.y - resized_glasses.rows / 2.0f);
183
184 overlay_with_alpha(result, resized_glasses, x, y);
185
186 return result;
187 }

192 cv::Mat apply_mask(const cv::Mat& aligned_face) {
193 if (!has_mask_) {
194 return aligned_face.clone();
195 }
196
197 cv::Mat result = aligned_face.clone();
198
199 // Mask should cover from nose tip to below mouth
200 cv::Point2f mask_center((ref_left_mouth_.x + ref_right_mouth_.x) / 2.0f,
201 (ref_nose_.y + ref_left_mouth_.y) / 2.0f);
202 float face_width = std::abs(ref_right_eye_.x - ref_left_eye_.x) * 2.0f;
203
204 // Scale mask
205 float target_width = face_width;
206 float scale = target_width / mask_template_.cols;
207
208 cv::Mat resized_mask;
209 cv::resize(mask_template_, resized_mask, cv::Size(), scale, scale, cv::INTER_LINEAR);
210
211 // Position mask
212 int x = static_cast<int>(mask_center.x - resized_mask.cols / 2.0f);
213 int y = static_cast<int>(ref_nose_.y - resized_mask.rows * 0.2f);
214
215 overlay_with_alpha(result, resized_mask, x, y);
216
217 return result;

223 cv::Mat apply_hat(const cv::Mat& aligned_face) {
224 if (!has_hat_) {
225 return aligned_face.clone();
226 }
227
228 cv::Mat result = aligned_face.clone();
229
230 // Hat should be above eyes, centered horizontally
231 float face_width = std::abs(ref_right_eye_.x - ref_left_eye_.x) * 2.5f;
232 float head_center_x = (ref_left_eye_.x + ref_right_eye_.x) / 2.0f;
233
234 // Scale hat
235 float target_width = face_width;
236 float scale = target_width / hat_template_.cols;
237
238 cv::Mat resized_hat;
239 cv::resize(hat_template_, resized_hat, cv::Size(), scale, scale, cv::INTER_LINEAR);
240
241 // Position hat above eyes
242 int x = static_cast<int>(head_center_x - resized_hat.cols / 2.0f);
243 int y = static_cast<int>(ref_left_eye_.y - resized_hat.rows * 0.9f);
244
245 overlay_with_alpha(result, resized_hat, x, y);
246
247 return result;

262 void overlay_with_alpha(cv::Mat& background, const cv::Mat& overlay, int x, int y) {
263 // Boundary check
264 int bg_width = background.cols;
265 int bg_height = background.rows;
266
267 // Calculate visible region
268 int x_start = std::max(0, x);
269 int y_start = std::max(0, y);
270 int x_end = std::min(bg_width, x + overlay.cols);
271 int y_end = std::min(bg_height, y + overlay.rows);
272
273 if (x_start >= x_end || y_start >= y_end) {
274 return; // No visible region
275 }
276
277 // Overlay region
278 int ov_x_start = x_start - x;
279 int ov_y_start = y_start - y;
280 int roi_width = x_end - x_start;
281 int roi_height = y_end - y_start;
282
283 cv::Mat bg_roi = background(cv::Rect(x_start, y_start, roi_width, roi_height));
284 cv::Mat ov_roi = overlay(cv::Rect(ov_x_start, ov_y_start, roi_width, roi_height));
285
286 // Ensure background is 3 channel
287 if (background.channels() == 1) {
288 cv::cvtColor(background, background, cv::COLOR_GRAY2BGR);
289 bg_roi = background(cv::Rect(x_start, y_start, roi_width, roi_height));
290 }
291
292 // Handle alpha blending
293 if (overlay.channels() == 4) {
294 // Split overlay into BGR and alpha
295 std::vector<cv::Mat> channels;
296 cv::split(ov_roi, channels);
297
298 cv::Mat alpha = channels[3];
299 cv::Mat overlay_bgr;
300 cv::merge(std::vector<cv::Mat>{channels[0], channels[1], channels[2]}, overlay_bgr);
301
302 // Normalize alpha to [0, 1]
303 cv::Mat alpha_f;
304 alpha.convertTo(alpha_f, CV_32F, 1.0 / 255.0);
305
306 // Blend each channel
307 for (int c = 0; c < 3; c++) {
308 for (int row = 0; row < roi_height; row++) {
309 for (int col = 0; col < roi_width; col++) {
310 float a = alpha_f.at<float>(row, col);
311 bg_roi.at<cv::Vec3b>(row, col)[c] = static_cast<uchar>(
312 a * overlay_bgr.at<cv::Vec3b>(row, col)[c] +
313 (1.0f - a) * bg_roi.at<cv::Vec3b>(row, col)[c]
314 );
315 }
316 }
317 }
318 } else {
319 // No alpha, just copy
320 ov_roi.copyTo(bg_roi);
321 }
322 }
