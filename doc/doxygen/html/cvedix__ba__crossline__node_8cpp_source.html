103  bool cvedix_ba_crossline_node::at_1_side_of_line(cvedix_objects::cvedix_point p, cvedix_objects::cvedix_line line) {
104  auto p1 = line.start;
105  auto p2 = line.end;
106 
107  if (p1.x == p2.x) {
108  return p.x < p1.x;
109  }
110 
111  if (p1.y == p2.y) {
112  return p.y < p1.y;
113  }
114 
115  if (p2.x < p1.x) {
116  auto tmp = p2;
117  p2 = p1;
118  p1 = tmp;
119  }
120 
121  int ret = (p2.y - p.y) * (p2.x - p1.x) - (p2.y - p1.y) * (p2.x - p.x);
122  return ret < 0;
123  }
