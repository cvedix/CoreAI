153  int cvedix_rknn_helper::set_input(int index, const cv::Mat& mat) {
154  if (!initialized || ctx == 0) {
155  CVEDIX_ERROR("[RKNN] Not initialized");
156  return -1;
157  }
158 
159  if (index < 0 || index >= io_num.n_input) {
160  CVEDIX_ERROR(cvedix_utils::string_format("[RKNN] Invalid input index: %d", index));
161  return -1;
162  }
163 
164  auto& input_attr = input_attrs[index];
165 
166  // Prepare input data based on model requirements
167  cv::Mat input_mat = mat.clone();
168 
169  // Convert BGR to RGB if needed (most RKNN models expect RGB)
170  if (input_attr.fmt == RKNN_TENSOR_NHWC) {
171  if (mat.channels() == 3) {
172  cv::cvtColor(input_mat, input_mat, cv::COLOR_BGR2RGB);
173  }
174  }
175 
176  // Resize if needed
177  int model_w = input_attr.dims[2]; // Usually width
178  int model_h = input_attr.dims[1]; // Usually height
179  if (input_mat.cols != model_w || input_mat.rows != model_h) {
180  cv::resize(input_mat, input_mat, cv::Size(model_w, model_h));
181  }
182 
183  // Normalize if needed (assuming uint8 input)
184  if (input_attr.type == RKNN_TENSOR_FLOAT32) {
185  input_mat.convertTo(input_mat, CV_32F, 1.0/255.0);
186  }
187 
188  // Prepare input structure
189  inputs[index].index = index;
190  inputs[index].type = input_attr.type;
191  inputs[index].fmt = input_attr.fmt;
192  inputs[index].size = input_mat.total() * input_mat.elemSize();
193  inputs[index].buf = input_mat.data;

195  return 0;

252  cv::Mat cvedix_rknn_helper::get_output_mat(int index, const std::vector<int>& shape) {
253  if (!initialized) {

256

259  if (data == nullptr) {

262 
263  // Create Mat from output data

265  if (shape.size() == 4) {

267  int total_size = 1;

269 
270  if (total_size == size) {
