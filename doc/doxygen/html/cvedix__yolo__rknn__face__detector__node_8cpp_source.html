148  void cvedix_yolo_rknn_face_detector_node::parse_yolov8_output(
149  float* output_data,
150  int output_size,
151  const cv::Size& frame_size,
152  std::vector<cv::Rect>& boxes,
153  std::vector<float>& scores) {
154 
155  // YoloV8 output format: [num_detections, 6] where 6 = [x_center, y_center, width, height, conf, class]
156  // Or: [1, num_boxes, 85] for standard YOLO format
157  // This implementation assumes output is flattened: [num_boxes * 6]
158 
159  boxes.clear();
160  scores.clear();
161 
162  // Assuming output format: [num_boxes, 6] where 6 = [x, y, w, h, conf, class]
163  // Or it could be [1, num_boxes, 85] for YOLO format with 80 classes
164  // For face detection, we typically have 1 class, so format might be [num_boxes, 6]
165 
166  int num_boxes = output_size / 6; // Assuming 6 values per box
167 
168  float scale_x = (float)frame_size.width / input_width;
169  float scale_y = (float)frame_size.height / input_height;
170 
171  for (int i = 0; i < num_boxes; i++) {
172  float* box_data = output_data + i * 6;
173 
174  float x_center = box_data[0];
175  float y_center = box_data[1];
176  float width = box_data[2];
177  float height = box_data[3];
178  float conf = box_data[4];
179  float cls = box_data[5];
180 
181  // Filter by confidence
182  if (conf < score_threshold) {
183  continue;
184  }
185 
186  // Convert from center format to top-left format
187  int x = (int)((x_center - width / 2) * scale_x);
188  int y = (int)((y_center - height / 2) * scale_y);
189  int w = (int)(width * scale_x);
190  int h = (int)(height * scale_y);
191 
192  // Clamp to image bounds
193  x = std::max(0, std::min(x, frame_size.width - 1));
194  y = std::max(0, std::min(y, frame_size.height - 1));
195  w = std::max(1, std::min(w, frame_size.width - x));
196  h = std::max(1, std::min(h, frame_size.height - y));
197 
198  boxes.push_back(cv::Rect(x, y, w, h));
199  scores.push_back(conf);
200  }
